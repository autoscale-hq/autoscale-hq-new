// Lead Processing Automation
export const processLead = async (payload) => {
  try {
    // 1. Parse incoming webhook
    const lead = validateData(payload);

    // 2. Enrich data via Clearbit
    const enriched = await enrichProfile(lead.email);

    // 3. Route to sales team in Slack
    if (enriched.companySize > 50) {
      await notifySlack("#enterprise-leads", enriched);
    }
    return { status: "success", routed: true };
  } catch (error) {
    await logError(error);
  }
};
